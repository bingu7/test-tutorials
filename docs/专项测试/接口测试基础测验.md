<script>
document.addEventListener('DOMContentLoaded', function() {
    const quizContainer = document.querySelector('.quiz-container');
    const quizItems = document.querySelectorAll('.quiz-item');
    const resultContainer = document.querySelector('.quiz-result');
    const scoreElement = document.querySelector('.quiz-score');
    const percentageElement = document.querySelector('.quiz-percentage');
    const feedbackElement = document.querySelector('.quiz-feedback');
    
    let score = 0;
    let answered = 0;
    
    quizItems.forEach((item, index) => {
        const options = item.querySelectorAll('input[type="radio"]');
        const explanation = item.querySelector('.quiz-explanation');
        
        options.forEach(option => {
            option.addEventListener('change', function() {
                if (this.value === item.dataset.correct) {
                    score++;
                    item.style.borderLeft = '4px solid #4CAF50';
                } else {
                    item.style.borderLeft = '4px solid #f44336';
                }
                explanation.style.display = 'block';
                answered++;
                
                if (answered === quizItems.length) {
                    showResult();
                }
            });
        });
    });
    
    function showResult() {
        scoreElement.textContent = score;
        percentageElement.textContent = Math.round((score / quizItems.length) * 100) + '%';
        
        let feedback = '';
        const percentage = (score / quizItems.length) * 100;
        
        if (percentage >= 90) {
            feedback = '🎉 优秀！你对接口测试掌握得非常好！';
        } else if (percentage >= 70) {
            feedback = '👍 良好！你对接口测试有较好的理解，建议复习错题。';
        } else if (percentage >= 60) {
            feedback = '📚 及格！建议重新学习接口测试教程。';
        } else {
            feedback = '💪 需要努力！请认真学习接口测试教程。';
        }
        
        feedbackElement.innerHTML = feedback;
        resultContainer.style.display = 'block';
        resultContainer.scrollIntoView({ behavior: 'smooth' });
    }
});
</script>
